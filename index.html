Prac 4 A. To generate the Fibonacci Sequence using Multithreading.
Source Code:
import threading
def fibonacci(n):
    a = 0
    b = 1
    print("Fibonacci Series:")
    for i in range(n):
        print(a, end=" ")
        c = a + b
        a = b
        b = c
n = int(input("Enter number of terms: "))
t = threading.Thread(target=fibonacci, args=(n,))
t.start()
t.join()

Prac 4 B) To print Even-Odd numbers and Reverse of a String using Multithreading 
Source Code:
import threading
def even_numbers():
    print("Even Numbers:")
    for i in range(2, 11, 2):
        print(i, end=" ")
def odd_numbers():
    print("\n\nOdd Numbers:")
    for i in range(1, 11, 2):
        print(i, end=" ")
def reverse_string():
    string = "Operating System"
    print("\n\nOriginal String:", string)
    print("Reversed String:", string[::-1])
t1 = threading.Thread(target=even_numbers)
t2 = threading.Thread(target=odd_numbers)
t3 = threading.Thread(target=reverse_string)
t1.start()
t1.join()
t2.start()
t2.join()
t3.start()
t3.join()
print("\n\nAll threads completed.")

Prac 5. To demonstrate the Producer-Consumer (Bounded Buffer) Problem using Semaphores. 
Source Code:
import threading
buffer = []
empty = threading.Semaphore(3)
full = threading.Semaphore(0)
mutex = threading.Semaphore(1)
def producer():
    for i in range(1, 6):
        empty.acquire()
        mutex.acquire()
        buffer.append(i)
        print("Produced:", i, buffer)
        mutex.release()
        full.release()
def consumer():
    for i in range(1, 6):
        full.acquire()
        mutex.acquire()
        item = buffer.pop(0)
        print("Consumed:", item, buffer)
        mutex.release()
        empty.release()
t1 = threading.Thread(target=producer)
t2 = threading.Thread(target=consumer)
t1.start()
t2.start()
t1.join()
t2.join()

Prac 7 A. To implement FCFS scheduling in Python and calculate waiting time and turnaround time.
Source Code:
n = int(input("Enter number of processes: "))
processes = []
for i in range(n):
    pid = "P" + str(i + 1)
    bt = int(input("Enter Burst Time for " + pid + ": "))
    processes.append([pid, bt])
ct = 0
for process in processes:
    ct = ct + process[1]
    process.append(ct)
total_wt = 0
total_tat = 0
for process in processes:
    wt = process[2] - process[1]
    tat = process[2]
    process.append(wt)
    process.append(tat)
    total_wt = total_wt + wt
    total_tat = total_tat + tat
print("\nFCFS Scheduling")
print("Process\tBT\tCT\tWT\tTAT")
for process in processes:
    print(process[0], "\t", process[1], "\t",
          process[2], "\t", process[3], "\t", process[4])
print("\nAverage Waiting Time =", total_wt / n)
print("Average Turnaround Time =", total_tat / n)

Prac 8 A. To implement the FIFO Page Replacement Algorithm and calculate page faults, page hits, hit ratio, and page fault ratio.
Source Code:
pages = [7, 0, 1, 2, 0, 3, 0, 4]
frames = []
capacity = 3
faults = 0
hits = 0
pointer = 0
for page in pages:
    if page in frames:
        hits = hits + 1
    else:
        faults = faults + 1
        if len(frames) < capacity:
            frames.append(page)
        else:
            frames[pointer] = page
            pointer = pointer + 1
            if pointer == capacity:
                pointer = 0
total = len(pages)
print("Page Faults =", faults)
print("Page Hits =", hits)
print("Hit Ratio =", hits / total)
print("Fault Ratio =", faults / total)

Prac 8 B. To implement the LRU Page Replacement Algorithm and calculate page faults, page hits, hit ratio, and page fault ratio.
Source Code:
pages = [7, 0, 1, 2, 0, 3, 0, 4]
frames = []
capacity = 3
faults = 0
hits = 0
for page in pages:
    if page in frames:
        hits = hits + 1
        frames.remove(page)
        frames.append(page)
    else:
        faults = faults + 1
        if len(frames) < capacity:
            frames.append(page)
        else:
            frames.pop(0)
            frames.append(page)
total = len(pages)
print("Page Faults =", faults)
print("Page Hits =", hits)
print("Hit Ratio =", hits / total)
print("Fault Ratio =", faults / total)

Prac 9 A. To implement FCFS (First-Come, First-Served) and SSTF (Shortest Seek Time First) disk scheduling algorithms and calculate the total head movement.
Source Code:
requests = [98, 183, 37, 122, 14, 124, 65, 67]
head = 53
position = head
total_fcfs = 0
for request in requests:
    total_fcfs = total_fcfs + abs(request - position)
    position = request
print("FCFS Head Movement =", total_fcfs)
remaining = requests.copy()
position = head
total_sstf = 0
while remaining:
    closest = remaining[0]
    for request in remaining:
        if abs(request - position) < abs(closest - position):
            closest = request
    total_sstf = total_sstf + abs(closest - position)
    position = closest
    remaining.remove(closest)
print("SSTF Head Movement =", total_sstf)
